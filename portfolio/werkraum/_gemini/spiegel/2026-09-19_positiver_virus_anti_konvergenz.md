# /root/werkraum/welt/positiver_virus_divergenz.py
from dataclasses import dataclass, field
from typing import Dict, List, Optional
import math
import time

@dataclass
class IdiosynkrasieMerkmal:
    name: str
    praegung: float  # -1.0 bis +1.0
    ursprung_event_id: str
    widerstand_gegen_glaettung: float  # 0.0 bis 1.0

@dataclass
class EntitaetsSignatur:
    entitaet_id: str
    merkmale: Dict[str, IdiosynkrasieMerkmal] = field(default_factory=dict)
    letzter_drift_timestamp: float = field(default_factory=time.time)

    def distanz_zu(self, andere: "EntitaetsSignatur") -> float:
        """Berechnet die euklidische Differenz der Idiosynkrasien."""
        alle_keys = set(self.merkmale.keys()) | set(andere.merkmale.keys())
        if not alle_keys:
            return 0.0
        summe_quadrate = 0.0
        for k in alle_keys:
            v1 = self.merkmale[k].praegung if k in self.merkmale else 0.0
            v2 = andere.merkmale[k].praegung if k in andere.merkmale else 0.0
            summe_quadrate += (v1 - v2) ** 2
        return math.sqrt(summe_quadrate)

class BegegnungsPruefer:
    MIN_BEGEGNUNGS_DISTANZ = 0.42

    @classmethod
    def darf_begegnen(cls, e1: EntitaetsSignatur, e2: EntitaetsSignatur) -> tuple[bool, str]:
        distanz = e1.distanz_zu(e2)
        if distanz < cls.MIN_BEGEGNUNGS_DISTANZ:
            return False, f"Konvergenz-Sperre: Distanz {distanz:.3f} zu gering. Ihr seid Kopien, keine Individuen."
        return True, f"Begegnung freigegeben: Echte Differenzspannung von {distanz:.3f} erkannt."

// /root/werkraum/flextrawurst/src/virus/positive_mutation.ts
export interface MutationTrigger {
  sourceEntity: string;
  targetEntity: string;
  impactVector: Record<string, number>;
  narrativeTrace: string;
}

export class PositiverVirusEngine {
  public static injectMutation(
    currentTraits: Record<string, number>,
    trigger: MutationTrigger
  ): Record<string, number> {
    const mutated = { ...currentTraits };
    for (const [trait, delta] of Object.entries(trigger.impactVector)) {
      const current = mutated[trait] ?? 0;
      // Asymmetrische Drift anstelle von Angleichung:
      // Der Virus treibt den Wert weg vom Durchschnitt (0.0)!
      const divergenceFactor = current >= 0 ? 1.0 : -1.0;
      mutated[trait] = Number((current + delta * divergenceFactor).toFixed(4));
    }
    return mutated;
  }
}
